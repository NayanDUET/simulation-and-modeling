
a1 = int(input("Enter a1"))
a2 = int(input("Enter a2"))
m1 = int(input("Enter m1"))

import matplotlib.pyplot as plt
import numpy as np
# Define x range
x = np.linspace(p - 10, p + 10, 100)

# Line equations
y1 = a1 + m1 * x
y2 = a2 + m2 * x

# Plot
plt.figure()

# Intersection point
plt.scatter(p, Q)
